def digest(data: bytes) -> str:
    return hashlib.sha256(data).hexdigest()


def octal_bytes(data: bytes) -> str:
    return "".join(f"\\0{byte:03o}" for byte in data)


def module_ok(remote: Remote, command: str, timeout: int = 90) -> str:
    rc, out = remote.module(command, timeout=timeout)
    print(out, end="" if out.endswith("\n") else "\n")
    if rc != 0 or "ADB_RESYNC_OK" not in out:
        raise RuntimeError(f"module command failed rc={rc}")
    return out


def upload(remote: Remote, path: str, data: bytes, expected: str) -> None:
    module_ok(
        remote,
        f"set -eu; test ! -e {path}; test ! -L {path}; "
        f"set -C; : > {path}; chmod 0600 {path}; echo CANDIDATE_CREATED={path}",
        timeout=45,
    )
    try:
        for offset in range(0, len(data), 240):
            chunk = octal_bytes(data[offset : offset + 240])
            module_ok(remote, f"set -eu; printf '%b' '{chunk}' >> {path}", timeout=45)
        module_ok(
            remote,
            f"set -eu; test \"$(sha256sum {path} | cut -d' ' -f1)\" = {expected}; "
            f"echo CANDIDATE_HASH_OK={path}",
            timeout=45,
        )
    except Exception:
        try:
            module_ok(remote, f"rm -f {path}; echo CANDIDATE_ROLLED_BACK={path}", timeout=30)
        except Exception:
            pass
        raise


def vendor_guard() -> str:
    return "\n".join(
        f'test "$(sha256sum {path} | cut -d\' \' -f1)" = {sha}'
        for path, sha in VENDOR.items()
    )


def rfs_guard() -> str:
    return r"""for p in /data/persist/rfs/shared /data/persist/hlos_rfs/shared; do
    test -d "$p"; test ! -L "$p"
    test "$(stat -c '%a:%U:%G' "$p")" = '777:system:system'
    test "$(ls -dZ "$p" | awk '{print $1}')" = 'u:r:tftp_server.datafile'
done"""

            "/tmp/dw_dms.success",
        ]
    )
    return f"""set -eu
echo FACTORY_PERSISTENCE_PREFLIGHT_BEGIN
grep -q "DISTRIB_TARGET='sdx75/generic'" /etc/openwrt_release
grep -q ' /data ubifs rw' /proc/mounts
grep -q ' /etc overlay rw' /proc/mounts
test "$(getenforce)" = Enforcing
test "$(cat /sys/class/remoteproc/remoteproc0/state)" = running
{vendor_guard()}
{rfs_guard()}
test -d /data/mbimd; test ! -L /data/mbimd
test -f /data/mbimd/disable_fcc_lock_flag; test ! -L /data/mbimd/disable_fcc_lock_flag

        list(TARGETS.values())
        + list(temp.values())
        + ["/etc/rc.d/S10fcc_bypass", "/etc/rc.d/S86dw_dms", "/data/dw_dms.last"]
    )

        f'test "$(sha256sum {CANDIDATES[name]} | cut -d\' \' -f1)" = {EXPECTED[name]}'
        for name in ("dms", "ctl", "fcc", "init")
    )

        f'test "$(sha256sum {temp[name]} | cut -d\' \' -f1)" = {EXPECTED[name]}'
        for name in ("dms", "ctl", "fcc", "init")

def qmi_read(remote: Remote, option: str, timeout: int = 35) -> tuple[int, str]:
    command = (
        "set -o pipefail; mkdir -p /var/lock; "
        f"timeout -k 3 25 flock {QMI_LOCK} qmicli -p -d /dev/mhi_QMI0 "
        f"--device-open-qmi {option} 2>&1"
    )
    return remote.sudo(command, timeout=timeout)


def main() -> int:
    artifacts = {
        "dms": DMS_SOURCE.read_bytes(),
        "ctl": (HERE / "dw_dms_ctl").read_bytes(),
        "fcc": (HERE / "fcc_bypass.init").read_bytes(),
        "init": (HERE / "dw_dms.init").read_bytes(),
    }
    for name, data in artifacts.items():
        if digest(data) != EXPECTED[name]:
            raise RuntimeError(f"local artifact hash mismatch: {name}")

    report: dict[str, object] = {
        "schema": "codex.dw5934e.factory-module-persistence-deployment.v1",
        "artifact_sha256": EXPECTED,
        "vendor_sha256": VENDOR,
        "host_dms_set_requests": 0,
        "dms_reset_requests": 0,
        "device_reboots": 0,
        "driver_actions": 0,
        "cold_boot_verified": False,
    }
    remote = Remote(read_sudo_password())
    uploaded: list[str] = []
    try:
        report["preflight"] = module_ok(remote, module_preflight(), timeout=70)

        for name in ("dms", "ctl", "fcc", "init"):
            upload(remote, CANDIDATES[name], artifacts[name], EXPECTED[name])
            uploaded.append(CANDIDATES[name])

        installer = installer_script()
        installer_sha = digest(installer)
        upload(remote, CANDIDATES["installer"], installer, installer_sha)
        uploaded.append(CANDIDATES["installer"])

        install_out = module_ok(
            remote,
            f"set -eu; test \"$(sha256sum {CANDIDATES['installer']} | cut -d' ' -f1)\" = {installer_sha}; "
            f"chmod 0700 {CANDIDATES['installer']}; {CANDIDATES['installer']}",
            timeout=120,
        )
        if "FACTORY_PERSISTENCE_INSTALL_COMMITTED" not in install_out:
            raise RuntimeError("install commit sentinel missing")
        report["install"] = install_out
        report["installer_sha256"] = installer_sha

        report["activation"] = activate_and_verify(remote)

        dms_records: list[dict[str, object]] = []
        for index in range(3):
            rc, out = qmi_read(remote, "--dms-get-operating-mode")
            print(f"=== DMS_POST_{index + 1} rc={rc} ===")
            print(out, end="" if out.endswith("\n") else "\n")
            mode = re.search(r"Mode: '([^']+)'", out)
            dms_records.append({"rc": rc, "mode": mode.group(1) if mode else None, "output": out})
            if rc != 0 or not mode or mode.group(1) != "online" or "HW restricted: 'no'" not in out:
                raise RuntimeError("DMS post-deployment verification failed")
            time.sleep(2)
        report["dms_post"] = dms_records

        rc, uim = qmi_read(remote, "--uim-get-card-status", timeout=45)
        print(f"=== UIM_CARD rc={rc} ===")
        print(uim, end="" if uim.endswith("\n") else "\n")
        report["uim"] = {"rc": rc, "output": uim}
        if rc != 0 or "Card state: 'present'" not in uim or "Card error: 'none'" not in uim:
            raise RuntimeError("SIM card is not ready")

        rc, nas = qmi_read(remote, "--nas-get-serving-system", timeout=45)
        print(f"=== NAS_SERVING rc={rc} ===")
        print(nas, end="" if nas.endswith("\n") else "\n")
        report["nas"] = {"rc": rc, "output": nas}

        rc, signal = qmi_read(remote, "--nas-get-signal-info", timeout=45)
        print(f"=== NAS_SIGNAL rc={rc} ===")
        print(signal, end="" if signal.endswith("\n") else "\n")
        report["signal"] = {"rc": rc, "output": signal}

        report["state"] = "installed_live_online_sim_readable"
        report["finished_unix"] = time.time()
        print("FACTORY_MODULE_PERSISTENCE_DEPLOYMENT=PASS")
        return 0
    finally:
        for path in uploaded:
            try:
                module_ok(remote, f"rm -f {path}; echo CANDIDATE_REMOVED={path}", timeout=30)
            except Exception:
                pass
        remote.close()
        out_path = HERE / "deployment_result_20260828.json"
        out_path.write_text(json.dumps(report, indent=2, sort_keys=True) + "\n", encoding="utf-8")
        print(f"RESULT_FILE={out_path}")


if __name__ == "__main__":
    try:
        raise SystemExit(main())
    except Exception as exc:
        print(f"FACTORY_MODULE_PERSISTENCE_DEPLOYMENT=FAIL:{type(exc).__name__}:{exc}", file=sys.stderr)
        raise
